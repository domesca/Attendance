async function submitAttendance(action) { const name = document.getElementById("name").value.trim(); if (!name) { alert("Please enter your name"); return; } navigator.geolocation.getCurrentPosition(async (pos) => { const ip = await getIP(); const form = document.createElement("form"); form.method = "POST"; form.action = "PASTE_YOUR_EXEC_URL_HERE"; form.target = "_blank"; const fields = { name: name, action: action, lat: pos.coords.latitude, lon: pos.coords.longitude, ip: ip, timestamp: new Date().toISOString() }; for (const key in fields) { const input = document.createElement("input"); input.type = "hidden"; input.name = key; input.value = fields[key]; form.appendChild(input); } document.body.appendChild(form); form.submit(); document.body.removeChild(form); alert("Attendance submitted"); }, () => { alert("Please allow location access."); }); }
